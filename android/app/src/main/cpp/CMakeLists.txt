cmake_minimum_required(VERSION 3.22.1)
project(thimvale LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Kotlin remains debuggable; the tensor kernels must not run at -O0 in development.
set(CMAKE_C_FLAGS_DEBUG "-g -O3")
set(CMAKE_CXX_FLAGS_DEBUG "-g -O3")
set(LLAMA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../Vendor/llama.cpp-b10830")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_ROOT}" llama-build)
add_library(thimvale SHARED inference.cpp)
target_include_directories(thimvale PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../Vendor")
target_link_libraries(thimvale llama android log)
target_link_options(thimvale PRIVATE "-Wl,-z,max-page-size=16384")
